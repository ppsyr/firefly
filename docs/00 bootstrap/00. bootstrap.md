            │   │         └─ except Exception as exc:
            │   │               └─ console.print(f"[red]Model switch failed: {exc}[/red]\n")
            │   │
            │   └─ continue（本轮不进 graph）
            │
            ├─ 【分支 B】普通输入
            │   ├─ renderer.render_user_input(prompt)
            │   ├─ intent_tree.detect_and_dispatch(prompt, runtime)
            │   │     └─ True → continue（不进 graph）
            │   │
			│   └─ 【流式研究】—— 一次完整运行
			│       │
			│       ├─ ctx = runtime.run_manager.create_run(...)
			│       │     └─ ...（创建 RunContext / RunRecord / RunJournal，落盘 record.json）
			│       │
			│       ├─ runtime.run_manager.mark_running(ctx.run_id)
			│       │     └─ replace(record, status=RUNNING) + journal.append("run.started")
			│       │
			│       ├─ config = _build_stream_config(runtime, ctx)
			│       │     └─ configurable: expert_mode / run_id / thread_id / journal /
			│       │        output_dir / plan_enabled / timezone / model
			│       │        + recursion_limit = max_loop_steps * graph_node_multiplier
			│       │
			│       ├─ config["configurable"]["skill_override"] = cli_state["skill_override"]
			│       │
			│       ├─ client = PoirotStreamClient(graph=runtime.leader_agent.graph, config=config)
			│       │
			│       ├─ renderer.state["round_t0"] = time.monotonic()
			│       ├─ renderer.state["model"] = _resolve_actual_model_name(...)
			│       │
			│       ├─ async for event in client.stream(prompt):     ← ★ 驱动 graph
			│       │   │
			│       │   │   client.stream(question)
			│       │   │   │
			│       │   │   ├─ 初始化 state = {messages:[HumanMessage(question)],
			│       │   │   │                user_input, research_question}
			│       │   │   ├─ 初始化去重集合：seen_ids / streamed_ids / seen_tool_call_ids /
			│       │   │   │                _internal_answer_ids / _answer_buffers / _answer_safe
			│       │   │   ├─ first_values_frame = True
			│       │   │   │
			│       │   │   └─ async for item in graph.astream(      ← 每拿到一个 chunk 就迭代一次
			│       │   │         state, config,
			│       │   │         stream_mode=["values","messages","custom"]):
			│       │   │       │
			│       │   │       │  ┌─────────────────────────────────────────────────────┐
			│       │   │       │  │ ★ graph 内部：LangGraph 跑图                          │
			│       │   │       │  └─────────────────────────────────────────────────────┘
			│       │   │       │
			│       │   │       │ 【图骨架】create_agent 定的固定骨架
			│       │   │       │   START → agent ⇄ tools → END
			│       │   │       │
			│       │   │       │ 【agent 节点】
			│       │   │       │   ├─ before_agent 钩子链
			│       │   │       │   │     1c → 6b → 11c → 13c
			│       │   │       │   │
			│       │   │       │   ├─ ReAct 循环（每轮）：
			│       │   │       │   │     ├─ before_model 钩子链
			│       │   │       │   │     │     1c → 2 → 3 → 5 → 6b → 9 → 10 → 11b → 11c → 13c
			│       │   │       │   │     ├─ wrap_model_call 洋葱
			│       │   │       │   │     │     1a → 1b → 1c → 13c → model.invoke()
			│       │   │       │   │     │     （POST 反序：13c → 1c → 1b → 1a）
			│       │   │       │   │     ├─ model.invoke() ← 真正调 LLM
			│       │   │       │   │     ├─ after_model 钩子链（反序）
			│       │   │       │   │     │     13d → 13c → 13b → 10 → 9 → 6b → 1a → 1c
			│       │   │       │   │     └─ 有 tool_calls？
			│       │   │       │   │           ├─ 否 → 退出循环
			│       │   │       │   │           └─ 是 → 跳 tools 节点
			│       │   │       │   │
			│       │   │       │   └─ after_agent 钩子链（反序）
			│       │   │       │         14 → 13c → 8 → 6b → 6a → 4 → 1c
			│       │   │       │
			│       │   │       │ 【tools 节点】
			│       │   │       │   └─ wrap_tool_call 洋葱（PRE 正序 / POST 反序）
			│       │   │       │         PRE：1c → 4 → 6b → 7 → 8 → 11a → 11c → 12 → 13a → 13b → tool.invoke()
			│       │   │       │         POST：tool.invoke() → 13b → 13a → 12 → 11c → 11a → 8 → 7 → 6b → 4 → 1c
			│       │   │       │
			│       │   │       │ 【流转】
			│       │   │       │   agent → 有 tool_calls？
			│       │   │       │     ├─ 是 → tools → 回到 agent（ReAct 循环）
			│       │   │       │     └─ 否 → END
			│       │   │       │   干预：
			│       │   │       │     after_model.jump_to="model" → 强制跳回 model
			│       │   │       │     wrap_tool_call.goto=END → 强制跳 END
			│       │   │       │
			│       │   │       │ 【每次状态更新 → 三种模式吐数据】
			│       │   │       ├─【custom】- 谁主动 writer() 发了什么，就吐什么 - 自定义事件
			│       │   │       │     └─ yield StreamEvent(type=chunk["type"], content, tool_name,
			│       │   │       │                      tool_args, tool_result, msg_id, skills?)
			│       │   │       │
			│       │   │       ├─【messages】每次 LLM 产生一个 token，吐一个 token 增量
			│       │   │       │     ├─ 解包 (msg_chunk, metadata)
			│       │   │       │     ├─ 过滤 metadata.tags 含 "internal_llm" → continue
			│       │   │       │     ├─ msg_id = msg_chunk.id
			│       │   │       │     │
			│       │   │       │     ├─ AIMessageChunk / AIMessage：
			│       │   │       │     │     ├─ reasoning_content → yield thinking
			│       │   │       │     │     │     （streamed_ids.add(msg_id)）
			│       │   │       │     │     ├─ content delta → selector 缓冲检测：
			│       │   │       │     │     │     ├─ msg_id ∈ _internal_answer_ids → 丢弃
			│       │   │       │     │     │     ├─ msg_id ∈ _answer_safe → yield answer
			│       │   │       │     │     │     ├─ 累积到 _answer_buffers[msg_id]
			│       │   │       │     │     │     │     ├─ _is_skills_selector_output(buf) → 丢弃 + 记 _internal
			│       │   │       │     │     │     │     ├─ 首字符非 {/` → 确认非 selector → flush answer
			│       │   │       │     │     │     │     ├─ len(buf) > 200 → 确认非 selector → flush answer
			│       │   │       │     │     │     │     └─ 否则继续缓冲
			│       │   │       │     │     │     └─ yield answer
			│       │   │       │     │     └─ tool_calls → 按 id 去重（seen_tool_call_ids）+ 跳过空 name
			│       │   │       │     │           └─ yield tool_start(tool_name, tool_args, msg_id)
			│       │   │       │     │
			│       │   │       │     └─ ToolMessage：
			│       │   │       │           ├─ streamed_ids.add(msg_id)
			│       │   │       │           └─ yield tool_end(tool_name, tool_result=truncate(200))
			│       │   │       │
			│       │   │       └─【values】每次图状态更新，吐一份完整 state 快照
			│       │   │             ├─ budget = governance.default.budget
			│       │   │             │     └─ 非零（window>0）→ yield budget_update
			│       │   │             ├─ sandbox = state.sandbox
			│       │   │             │     └─ 有 sandbox_id → yield sandbox_update
			│       │   │             ├─ messages = state.messages
			│       │   │             ├─ first_values_frame：
			│       │   │             │     └─ 预填 seen_ids / streamed_ids（跳过 checkpoint 恢复的旧消息）
			│       │   │             │        + first_values_frame = False + continue
			│       │   │             └─ 逐消息去重（seen_ids / streamed_ids）：
			│       │   │                   ├─ 已 streamed → skip
			│       │   │                   ├─ AIMessage（非 streaming 模型）：
			│       │   │                   │     ├─ content 非 selector → yield answer
			│       │   │                   │     └─ tool_calls → 去重 → yield tool_start
			│       │   │                   └─ ToolMessage → yield tool_end
			│       │   │
			│       │   ├─ flush 残留 _answer_buffers（未决 msg_id，非 selector → yield answer）
			│       │   └─ yield done
			│       │
			│       └─ renderer.render(event)     ← 逐事件渲染
			│             ├─ thinking      → 累积 → 切出时 flush "+ Thought: Xms"
			│             ├─ answer        → 累积 full_answer
			│             ├─ tool_start    → 开 Live spinner
			│             ├─ tool_end      → 停 spinner + ✓ 摘要 + 存 tool_results
			│             ├─ budget_update → _update_budget → cli_state（toolbar 刷新）
			│             ├─ sandbox_update → 透传
			│             ├─ done          → Markdown 渲染 + 耗时尾行 + 分隔线 + 清 state
			│             └─ error         → 红字
			│
            │
            ├─ runtime.run_manager.mark_success(ctx.run_id)
            │   └─ replace(record, status=SUCCESS, finished_at, total_tokens)
            │       + journal.append("run.finished", usage)
            │
            ├─ console.print(run_id / events_path)
            │
            └─ 异常分支
                ├─ KeyboardInterrupt / CancelledError
                │     └─ mark_failed(ctx.run_id, "interrupted")
                └─ Exception
                      ├─ renderer._stop_spinner()
                      └─ mark_failed(ctx.run_id, str(exc))
```

**装配阶段（bootstrap_runtime，同步）**
```
load_config → AppConfig
  + ModelRouter（provider 链）
  + CapabilityRegistry（模型/工具/reporter/artifact/sandbox/skill/
                        specialist/subagent/memory 九类能力）
  + RunManager
  + make_lead_agent → LeaderAgent(graph = create_agent(
      model       ← FallbackChatModel（researcher 链）
      tools       ← builtin + registry + specialist_tools
      middleware  ← 20 项（按固定顺序）
      system_prompt ← 条件拼装
      state_schema  ← ThreadState
      checkpointer  ← InMemorySaver 单例
    ))
  = AppRuntime
```

**运行阶段（主循环 → 一次研究）**
```
用户输入 → intent 检测（未命中）→ create_run → mark_running
  → PoirotStreamClient.stream(question)
      → graph.astream(stream_mode=["values","messages","custom"])
          → LangGraph 跑图：
              before_model 钩子链（skill 注入 / memory 召回 / sandbox 绑定）
              → model 调用（FallbackChatModel → 厂商 SDK）
              → 工具调用（ReAct 循环 + Orchestration 派活 + Skill 打点）
              → after_model / after_agent 钩子链（memory 沉淀 / skill 评估 / L2 触发 / 报告合成）
          → 逐事件 yield StreamEvent
      → renderer.render(event) 逐事件渲染
  → mark_success / mark_failed
